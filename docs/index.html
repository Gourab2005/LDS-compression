Physical Deployment Topology

What Lives on the Edge Device vs. Receiver Server

A production IoT deployment does not clone git repos or execute build tools on the edge. The system is split strictly into sender and receiver responsibilities.

Implementation Guide

Multi-Language Minimal Integration

Both engines run as persistent subprocesses over OS pipes. Switch between real-time streaming and batched transport block modes below.

Where and How to Configure Batch Size:

In Batched Block mode, you control the batch size directly in your code via two explicit placeholders at the top of each script, or via the command line:

1. BATCH_TARGET_BYTES
Byte threshold. Change to 1024 * 1024 (1 MB for Cloud/Edge Gateway) or 64 * 1024 (64 KB for MCUs / LoRaWAN).
2. MAX_FLUSH_LINES
Line count threshold. Change to 500 or 1000 lines. Forces a block flush when log rate is low.
3. CLI: --batch-size=<bytes>
Pass directly to engine: --batch-size=1048576 (1 MB) or --batch-size=65536 (64 KB).

# 1. CONFIGURE BATCH SIZE HERE (Adjust to your requirements)

BATCH_TARGET_BYTES = 1024 * 1024  # 1 MB block target (e.g. 64*1024 for 64 KB MCU)

1 Batched Block Transport

Compresses a collection of log records into one single binary block.

2 Batched Block Compression Code
const { spawn } = require('child_process');
const readline = require('readline');

// -------------------------------------------------------------
// 1. CONFIGURE BATCH SIZE HERE (Adjust to your requirements)
// -------------------------------------------------------------
const BATCH_TARGET_BYTES = 1024 * 1024; // 1 MB target (use 64*1024 for 64 KB block)
const MAX_FLUSH_LINES    = 1000;        // Force flush if line count reached first

// 2. Launch engine with configured batch size
const batchEncoder = spawn('./bin/lds_encoder_engine.exe', ['--mode=batch', `--batch-size=${BATCH_TARGET_BYTES}`]);

const rl = readline.createInterface({ input: batchEncoder.stdout });

function compressBlock(logLinesArray) {
  return new Promise((resolve) => {
    rl.once('line', (res) => resolve(JSON.parse(res)));
    batchEncoder.stdin.write(logLinesArray.join('\n') + '\n---END_BLOCK---\n');
  });
}

// 3. Dispatch array of buffered records as ONE payload
(async () => {
  const records = Array.from({ length: 1000 }, (_, i) => `2026-10-05 02:00:01.${i} INFO BMS pack=HV01 temp=42C`);
  const block = await compressBlock(records);
  console.log(`Batched ${records.length} logs: ${block.raw_bytes}B -> ${block.comp_bytes}B (${block.saving_pct}% saved)`);
})();

package main

import (
    "bufio"
    "encoding/json"
    "fmt"
    "os/exec"
    "strings"
)

// -------------------------------------------------------------
// 1. CONFIGURE BATCH SIZE HERE (Adjust to your requirements)
// -------------------------------------------------------------
const BatchTargetBytes = 1024 * 1024 // 1 MB target (use 64*1024 for 64 KB MCU)
const MaxFlushLines    = 1000        // Force flush if line count reached first

type BatchResult struct {
    RawBytes  int     `json:"raw_bytes"`
    CompBytes int     `json:"comp_bytes"`
    Savings   float64 `json:"saving_pct"`
}

func main() {
    // 2. Launch engine with configured batch size
    cmd := exec.Command("./bin/lds_encoder_engine.exe", "--mode=batch", fmt.Sprintf("--batch-size=%d", BatchTargetBytes))
    stdin, _ := cmd.StdinPipe()
    stdout, _ := cmd.StdoutPipe()
    scanner := bufio.NewScanner(stdout)
    cmd.Start()

    // 3. Buffer lines up to MaxFlushLines or BatchTargetBytes
    var buf strings.Builder
    for i := 0; i < 500; i++ {
        buf.WriteString(fmt.Sprintf("2026-10-05 03:00:01.%03d INFO SENSOR temp=24.5C\n", i))
    }
    buf.WriteString("---END_BLOCK---\n")

    stdin.Write([]byte(buf.String()))
    if scanner.Scan() {
        var res BatchResult
        json.Unmarshal(scanner.Bytes(), &res)
        fmt.Printf("Batch Block: %dB -> %dB (%.1f%% saved)\n", res.RawBytes, res.CompBytes, res.Savings)
    }
}

1 Direct Batched Block API

Calls lds_encode_block() to pack an array of log lines into one binary packet.

2 Batched Block API Code
#include "lds.h"
#include <stdio.h>
#include <stdlib.h>

// -------------------------------------------------------------
// 1. CONFIGURE BATCH SIZE HERE (MCU SRAM / Transport MTU target)
// -------------------------------------------------------------
#define BATCH_TARGET_BYTES  (64 * 1024)  // 64 KB transport block size
#define BATCH_MAX_LINES     500          // Max lines to accumulate in buffer

void send_telemetry_batch(const char** log_lines_array, size_t line_count) {
    static LDSEncoder* enc = NULL;
    if (!enc) enc = lds_encoder_create();

    // Guard line count against configured batch limit
    if (line_count > BATCH_MAX_LINES) {
        line_count = BATCH_MAX_LINES;
    }

    // 2. Compress array of lines into ONE single binary transport block
    size_t block_bytes = 0;
    uint8_t* block_pkt = lds_encode_block(enc, log_lines_array, line_count, &block_bytes);

    // 3. Transmit single block over cellular HTTP POST / LoRaWAN frame
    printf("Batched %zu lines into %zu bytes single transport block\n", line_count, block_bytes);

    // 4. Free block buffer
    free(block_pkt);
}

Production Protocol Specification

Stream Synchronization & Auto-Recovery User Manual

A complete guide explaining how stateful log stream compression synchronizes across networks, how to handle out-of-order packets, and how to recover automatically with zero data corruption.

1. Sequence Tagging (seq)
Sender increments a monotonic counter (seq: 1, 2, 3...) per packet. Enables the receiver to detect missed, delayed, or duplicated frames instantly.
2. CRC32 Frame Integrity
Every frame carries a 32-bit checksum. The receiver validates the CRC before decoding, dropping corrupted bit-flipped packets before they reach compression tables.
3. 409 RESYNC Handshake
If the receiver restarts mid-stream, it returns HTTP 409 RESYNC_REQUIRED. The sender auto-resets its encoder and re-sends as Seq #1 without manual restarts.
4. In-Band Reorder Buffer
If packets arrive out of order (e.g. #1, #3, #2), the receiver buffers #3 in memory until #2 arrives, then drains both in exact order with zero template loss.
Real-World Synchronization & Recovery Life-Cycle 
[EDGE DEVICE (SENDER)]                                         [CLOUD RECEIVER (INGESTION)]
   │                                                                    │
   ├──► 1. Normal Stream: POST { seq: 10, crc32: "3a4f", b64: "..." } ──► Recv #10, Verifies CRC ✓, Decodes bit-exact
   │                                                                    │
   │  [Server Restart / Maintenance Event Occurs]                       └──► Server restarts (Clean template memory)
   │                                                                    │
   ├──► 2. Out-of-Sync Packet: POST { seq: 11, ... } ────────────────► Detects Seq #11 without Seq #1..#10
   │                                                                    │
   │◄── 3. Handshake Signal: HTTP 409 RESYNC_REQUIRED ─────────────────┘
   │
   ├──► 4. Auto-Reset: Writes "RESET\n" to lds_encoder_engine.exe
   │    Encoder clears dictionary, starts new session at Seq #1
   │
   └──► 5. Re-send: POST { seq: 1, crc32: "9c1a", b64: "[OP_SYNC...]" } ─► Server registers new templates, stream 100% restored!

1 Edge Sender (Node.js) — CRC32 & Auto-Resync
const http = require('http');
const { spawn } = require('child_process');
const readline = require('readline');
const { crc32 } = require('zlib');

const encoder = spawn('./bin/lds_encoder_engine.exe');
const encoderRl = readline.createInterface({ input: encoder.stdout });
let seq = 0, sessionId = 'session_' + Date.now();

function resetEncoder() {
  return new Promise((resolve) => {
    encoderRl.once('line', () => resolve());
    encoder.stdin.write('RESET\n');
  });
}

async function sendCompressedLog(logLine) {
  seq++;
  // 1. Compress with native LDS engine
  const enc = await new Promise(r => {
    encoderRl.once('line', l => r(JSON.parse(l)));
    encoder.stdin.write(logLine.trim() + '\n');
  });

  // 2. Compute CRC32 checksum
  const payloadCrc = crc32(Buffer.from(enc.b64)).toString(16).padStart(8, '0');

  // 3. Post to receiver with 409 RESYNC handler
  const payload = JSON.stringify({ sessionId, seq, b64: enc.b64, crc32: payloadCrc });
  const req = http.request('http://localhost:5000/api/logs/ingest', {
    method: 'POST',
    headers: { 'Content-Type': 'application/json' }
  }, async (res) => {
    if (res.statusCode === 409) {
      // Server rebooted -> Auto-reset and re-send as Seq #1!
      console.log('[RESYNC] Server reboot detected. Resetting encoder state...');
      await resetEncoder();
      seq = 0;
      sessionId = 'session_' + Date.now();
      await sendCompressedLog(logLine);
    }
  });
  req.write(payload); req.end();
}
2 Cloud Receiver (Node.js) — Reorder Buffer & In-Band Reset
const http = require('http');
const { spawn } = require('child_process');
const readline = require('readline');
const { crc32 } = require('zlib');

const decoder = spawn('./bin/lds_decoder_engine.exe');
const decoderRl = readline.createInterface({ input: decoder.stdout });
let expectedSeq = 1, currentSession = null;
const reorderBuffer = new Map();

http.createServer((req, res) => {
  let body = '';
  req.on('data', c => body += c);
  req.on('end', async () => {
    const p = JSON.parse(body);

    // 1. Validate CRC32 Checksum
    if (p.crc32 && crc32(Buffer.from(p.b64)).toString(16).padStart(8,'0') !== p.crc32.toLowerCase()) {
      res.writeHead(400); res.end(JSON.stringify({ error: 'CRC32_CORRUPT' })); return;
    }

    // 2. Check for Fresh Session or Reboot Resync
    if (p.seq === 1) {
      decoder.stdin.write('RESET\n'); // In-band reset
      expectedSeq = 1; currentSession = p.sessionId; reorderBuffer.clear();
    } else if (currentSession === null) {
      // Receiver just booted mid-stream -> Send 409 RESYNC
      res.writeHead(409); res.end(JSON.stringify({ error: 'RESYNC_REQUIRED' })); return;
    }

    // 3. Out-of-Order Packet Buffering
    if (p.seq > expectedSeq) {
      reorderBuffer.set(p.seq, p);

func ingestHandler(w http.ResponseWriter, r *http.Request) {
    var p LogPacket
    json.NewDecoder(r.Body).Decode(&p)

    // 1. Validate CRC32 Checksum
    calcCRC := fmt.Sprintf("%08x", crc32.ChecksumIEEE([]byte(p.B64)))
    if p.CRC32 != "" && p.CRC32 != calcCRC {
        http.Error(w, `{"error":"CRC32_CORRUPT"}`, http.StatusBadRequest); return
    }

    mu.Lock()
    defer mu.Unlock()

    // 2. Check for Fresh Session / In-Band Reset
    if p.Seq == 1 {
        decIn.Write([]byte("RESET\n"))
        decReader.ReadString('\n')
        expectedSeq = 1; curSession = p.SessionID; reorderBuf = make(map[uint64]string)
    } else if curSession == "" {
        // Server rebooted mid-stream -> Send 409 RESYNC
        http.Error(w, `{"error":"RESYNC_REQUIRED"}`, http.StatusConflict); return
    }

    // 3. Reorder Buffering for Out-of-Order Packets
    if p.Seq > expectedSeq {
        reorderBuf[p.Seq] = p.B64
        w.WriteHeader(http.StatusAccepted); w.Write([]byte(`{"status":"BUFFERED"}`)); return
    }

    // 4. In-Order Decode & Drain Reorder Buffer
    decodeOne(p.B64); expectedSeq++
    for {
        if nextB64, ok := reorderBuf[expectedSeq]; ok {
            decodeOne(nextB64)
            delete(reorderBuf, expectedSeq)
            expectedSeq++
        } else { break }
    }
    w.Write([]byte(`{"status":"ACK"}`))
}

use actix_web::{web, App, HttpResponse, HttpServer, Responder};
use std::collections::BTreeMap;
use std::sync::Mutex;
use std::io::{BufRead, BufReader, Write};
use std::process::{Command, Stdio};

struct ServerState {
    expected_seq: u64,
    session_id: Option<String>,
    reorder_buffer: BTreeMap<u64, String>,
    dec_stdin: std::process::ChildStdin,
    dec_reader: BufReader<std::process::ChildStdout>,
}

async fn ingest(data: web::Json<serde_json::Value>, state: web::Data<Mutex<ServerState>>) -> impl Responder {
    let mut s = state.lock().unwrap();
    let seq = data["seq"].as_u64().unwrap_or(0);
    let b64 = data["b64"].as_str().unwrap_or("").to_string();

    // 1. Handle New Session / Reset
    if seq == 1 {
        writeln!(s.dec_stdin, "RESET").unwrap();
        let mut ack = String::new(); s.dec_reader.read_line(&mut ack).unwrap();
        s.expected_seq = 1; s.reorder_buffer.clear();
        s.session_id = Some(data["sessionId"].as_str().unwrap_or("").to_string());
    } else if s.session_id.is_none() {
        return HttpResponse::Conflict().json(serde_json::json!({"error":"RESYNC_REQUIRED"}));
    }

    // 2. Buffer Out-of-Order Packets
    if seq > s.expected_seq {
        s.reorder_buffer.insert(seq, b64);
        return HttpResponse::Accepted().json(serde_json::json!({"status":"BUFFERED"}));
    }

    // 3. In-Order Decode & Drain
    writeln!(s.dec_stdin, "{}", b64).unwrap();
    let mut dec_out = String::new(); s.dec_reader.read_line(&mut dec_out).unwrap();
    s.expected_seq += 1;

    while let Some(buffered_b64) = s.reorder_buffer.remove(&s.expected_seq) {
        writeln!(s.dec_stdin, "{}", buffered_b64).unwrap();
        let mut line = String::new(); s.dec_reader.read_line(&mut line).unwrap();
        s.expected_seq += 1;
    }

    HttpResponse::Ok().json(serde_json::json!({"status":"ACK"}))
}

void process_incoming_packet(const uint8_t* frame, size_t total_len) {
    if (!decoder) decoder = lds_decoder_create();
    if (total_len < 10 || frame[0] != 0x4C || frame[1] != 0x44) return;

    uint32_t pkt_seq, pkt_crc;
    memcpy(&pkt_seq, &frame[2], 4);

        lds_decoder_reset(decoder);
        expected_seq = 1;

    char* restored_log = lds_decode_line(decoder, payload, payload_len);

        printf("[INGEST] Seq #%u: %s\n", pkt_seq, restored_log);
        free(restored_log);
        expected_seq++;
    }

#include "lds.h"
#include "cmsis_os.h"
#include "lwip/sockets.h"

static LDSEncoder* mcu_encoder = NULL;
static uint32_t packet_seq = 0;

void LDS_Telemetry_Task(void *argument) {
    // Pre-allocate deterministic memory pool once at boot
    mcu_encoder = lds_encoder_create();
    int sock = connect_to_cloud_receiver("receiver.cloud.internal", 5000);

    for (;;) {
        char sensor_log[128];
        osMessageQueueGet(telemetryQueueHandle, sensor_log, NULL, osWaitForever);

        packet_seq++;
        size_t comp_len = 0;
        uint8_t* comp_data = lds_encode_line(mcu_encoder, sensor_log, &comp_len);

        // Construct 10-byte framing header: [0x4C 0x44] [Seq 4B] [CRC32 4B]
        uint8_t tx_buf[10 + comp_len];
        tx_buf[0] = 0x4C; tx_buf[1] = 0x44;
        memcpy(&tx_buf[2], &packet_seq, 4);
        uint32_t crc = lds_crc32(comp_data, comp_len);
        memcpy(&tx_buf[6], &crc, 4);
        memcpy(&tx_buf[10], comp_data, comp_len);

        int sent = send(sock, tx_buf, 10 + comp_len, 0);
        if (sent < 0) {
            // Socket reconnected -> Reset encoder dictionary
            lds_encoder_reset(mcu_encoder);
            packet_seq = 0;
            sock = connect_to_cloud_receiver("receiver.cloud.internal", 5000);
        }
        free(comp_data);
    }
}
